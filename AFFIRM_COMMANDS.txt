# ============================================================================
# AFFIRM STAGING QUERY - Read from AWS S3, Write to Local Iceberg
# ============================================================================

# STEP 0: GET FRESH AWS CREDENTIALS FIRST!
# Open ~/.aws/credentials and copy the CURRENT values:
#   - aws_access_key_id
#   - aws_secret_access_key  
#   - aws_session_token
#
# Session tokens expire every few hours, so refresh them if you get ExpiredToken errors!

# STEP 1: Compile the staging query
docker-compose exec -T chronon bash -c "cd /srv/chronon && python3 /srv/chronon-python/src/ai/chronon/repo/compile.py --conf staging_queries/affirm/paybright_repayment_data_v1_0_sq.py" <<< "y"

# STEP 2: Copy compiled config to spark-master
docker cp python/test/sample/compiled/staging_queries/affirm/paybright_repayment_data_v1_0_sq.v1__1 zipline-chronon-spark-master-1:/tmp/

# STEP 3: Run with FRESH AWS credentials
# Replace YOUR_AWS_ACCESS_KEY_ID, YOUR_AWS_SECRET_ACCESS_KEY, YOUR_AWS_SESSION_TOKEN
docker-compose exec -u root spark-master bash -c "/opt/spark/bin/spark-submit \
  --master local[*] \
  --conf spark.hadoop.fs.s3a.access.key=YOUR_AWS_ACCESS_KEY_ID \
  --conf spark.hadoop.fs.s3a.secret.key=YOUR_AWS_SECRET_ACCESS_KEY \
  --conf spark.hadoop.fs.s3a.session.token=YOUR_AWS_SESSION_TOKEN \
  --conf spark.hadoop.fs.s3a.path.style.access=false \
  --conf spark.hadoop.fs.s3a.impl=org.apache.hadoop.fs.s3a.S3AFileSystem \
  --conf spark.sql.catalog.spark_catalog=org.apache.iceberg.spark.SparkCatalog \
  --conf spark.sql.catalog.spark_catalog.type=hadoop \
  --conf spark.sql.catalog.spark_catalog.warehouse=file:///tmp/chronon-warehouse \ #this needs to be directed at an actual warehouse for avtual writing
  --packages org.apache.hadoop:hadoop-aws:3.3.4,org.apache.iceberg:iceberg-spark-runtime-3.5_2.12:1.5.0 \
  --class ai.chronon.spark.Driver \
  /tmp/chronon-spark.jar \
  staging-query-backfill \
  --conf-path /tmp/paybright_repayment_data_v1_0_sq.v1__1 \
  --end-date 2025-01-15"

# ============================================================================
# WHAT THIS DOES:
# - Reads: s3a://affirm-risk-sherlock-ca/feature-store/paybright_repayment_data/v1/*/*/*/out/
# - Writes: file:///tmp/chronon-warehouse (local Iceberg table on spark-master)
# - No MinIO conflicts! AWS S3 for input, local disk for output.
# ============================================================================
